# 导入镜像
docker load -i dify-web.tar

# 修改 docker-compose.yml 文件
``` yml
services:
  # ...历史配置
  difyWeb:
    image: dify-web:v0.1.22 # 这个版本号运行一下 docker images  查看一下
    container_name: dify-web
    ports:
      - "8015:3000"
    volumes:
      - ./html/dify:/app/web:ro
    networks:
      - my_network
```

# nginx 配置修改
``` conf
    server {

        # 开启 gzip 压缩
        gzip_static on; # 优先使用预压缩的 .gz 文件
        gzip on;
        gzip_min_length 1k;
        gzip_buffers 4 16k;
        gzip_comp_level 2;
        gzip_types text/plain text/css application/json application/javascript text/xml application/xml application/xml+rss text/javascript;
    
        # ...历史配置

        # dify 前端转发
        location /dify {
            proxy_pass http://8.130.142.178:8015;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }

        # dify 接口代理
        location /difyApi {
            rewrite ^/difyApi/(.*)$ /$1 break;
            proxy_pass http://8.130.142.178:8014;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }

        # dify 原接口代理，兼容处理
        location /console {
            proxy_pass http://8.130.142.178:8014;
            proxy_set_header Host $host;
            proxy_set_header X-Real-IP $remote_addr;
            proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
            proxy_set_header X-Forwarded-Proto $scheme;
            proxy_http_version 1.1;
            proxy_set_header Upgrade $http_upgrade;
            proxy_set_header Connection "upgrade";
        }
    }
```
